38   def find_icon_file():
39       from utils.path_helper import get_app_root
40       app_root = get_app_root()
41       candidates = [
42           os.path.join(app_root, 'resources', 'app_icon.ico'),
43           os.path.join(app_root, 'app_icon.ico'),
44           os.path.join(app_root, 'icon.ico'),
45       ]
46       for path in candidates:
47           if os.path.exists(path):
48               return path
49       return None

288      def load_registry(self, filters=None):
289          print("MainWindow.load_registry() вызван")
290          if filters is None:
291              rows = self.skzi_service.get_all_active()
292          else:
293              rows = self.skzi_service.get_all_active_filtered(filters)
294  
295          self.reg_table.setSortingEnabled(False)
296          self.reg_table.setRowCount(0)
297  
298          for i, row in enumerate(rows):
299              self.reg_table.insertRow(i)
300              chk = QTableWidgetItem()
301              chk.setFlags(Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsEnabled)
302              chk.setCheckState(Qt.CheckState.Unchecked)
303              self.reg_table.setItem(i, 0, chk)
304  
305              self.reg_table.setItem(i, 1, QTableWidgetItem(str(row['id'])))
306              self.reg_table.setItem(i, 2, QTableWidgetItem(row['fio']))
307              self.reg_table.setItem(i, 3, QTableWidgetItem(row['arm_type'] or ""))
308              self.reg_table.setItem(i, 4, QTableWidgetItem(row['arm_serial'] or ""))
309              self.reg_table.setItem(i, 5, QTableWidgetItem(row['skzi_name']))
310              self.reg_table.setItem(i, 6, QTableWidgetItem(row['skzi_number']))
311              self.reg_table.setItem(i, 7, QTableWidgetItem(row['media_type'] or ""))
312              self.reg_table.setItem(i, 8, QTableWidgetItem(row['media_number'] or ""))
313              self.reg_table.setItem(i, 9, QTableWidgetItem(row['cabinet_number'] or ""))
314  
315              # Преобразуем даты из ISO в формат dd.MM.yyyy для отображения
316              install_date = row['install_date']
317              if install_date:
318                  try:
319                      qdate = QDate.fromString(install_date, "yyyy-MM-dd")
320                      display_install = qdate.toString("dd.MM.yyyy")
321                  except:
322                      display_install = install_date
323              else:
324                  display_install = ""
325              self.reg_table.setItem(i, 10, QTableWidgetItem(display_install))
326  
327              expiry_date = row['expiry_date']
328              if expiry_date:
329                  try:
330                      qdate = QDate.fromString(expiry_date, "yyyy-MM-dd")
331                      display_expiry = qdate.toString("dd.MM.yyyy")
332                  except:
333                      display_expiry = expiry_date
334              else:
335                  display_expiry = ""
336              self.reg_table.setItem(i, 11, QTableWidgetItem(display_expiry))
337  
338              self.reg_table.setItem(i, 12, QTableWidgetItem(row['status']))
339  
340          self.reg_table.setSortingEnabled(True)
341          self.filter_registry()
342          self.reg_table.resizeColumnsToContents()
343          header = self.reg_table.horizontalHeader()
344          header.setStretchLastSection(True)
345          self.reg_table.viewport().update()
346          print("Таблица реестра обновлена")
347  
348      def get_selected_skzi_ids(self):
349          ids = []
350          for i in range(self.reg_table.rowCount()):
351              chk = self.reg_table.item(i, 0)
352              if chk and chk.checkState() == Qt.CheckState.Checked:
353                  id_item = self.reg_table.item(i, 1)
354                  if id_item:
355                      ids.append(int(id_item.text()))
356          return ids

358      def edit_selected_skzi(self):
359          ids = self.get_selected_skzi_ids()
360          if len(ids) == 0:
361              QMessageBox.warning(self, "Внимание", "Не выбрано ни одной записи.")
362              return
363          if len(ids) > 1:
364              QMessageBox.warning(self, "Внимание", "Выберите только одну запись для редактирования.")
365              return
366          dlg = SkziForm(self.user, self, record_id=ids[0])
367          if dlg.exec():
368              self.load_registry()

370      def mass_destroy_selected(self):
371          ids = self.get_selected_skzi_ids()
372          if not ids:
373              QMessageBox.warning(self, "Внимание", "Не выбрано ни одной записи.")
374              return
375          dlg = MassDestructionDialog(self)
376          if dlg.exec():
377              data = dlg.get_data()
378              if not data['act']:
379                  QMessageBox.warning(self, "Ошибка", "Введите номер акта.")
380                  return
381              if not data['withdrawer']:
382                  data['withdrawer'] = self.user['username']
383              reply = QMessageBox.question(self, "Подтверждение",
384                                           f"Вы действительно хотите уничтожить {len(ids)} записей?\nАкт №{data['act']} от {data['date']}",
385                                           QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No)
386              if reply == QMessageBox.StandardButton.Yes:
387                  self.skzi_service.mass_mark_destroyed(ids, data['date'], data['act'], data['withdrawer'], self.user['username'])
388                  QMessageBox.information(self, "Успех", f"{len(ids)} записей помечены как уничтоженные.")
389                  self.load_registry()

428      def filter_registry(self):
429          text = self.reg_search.text().lower()
430          for i in range(self.reg_table.rowCount()):
431              match = False
432              for j in range(self.reg_table.columnCount()):
433                  item = self.reg_table.item(i, j)
434                  if item and text in item.text().lower():
435                      match = True
436                      break
437              self.reg_table.setRowHidden(i, not match)

519          dialog = QDialog(self)
520          dialog.setWindowTitle(title)
521          layout = QVBoxLayout(dialog)
522  
523          layout.addWidget(QLabel("Начальная дата:"))
524          start_date = QDateEdit(calendarPopup=True)
525          start_date.setDate(QDate.currentDate().addMonths(-1))
526          layout.addWidget(start_date)
527  
528          layout.addWidget(QLabel("Конечная дата:"))
529          end_date = QDateEdit(calendarPopup=True)
530          end_date.setDate(QDate.currentDate())
531          layout.addWidget(end_date)
532  
533          btn_layout = QHBoxLayout()
534          btn_ok = QPushButton("Экспорт")
535          btn_ok.clicked.connect(lambda: self.do_export(dialog, start_date.date(), end_date.date(), export_func, default_filename))
536          btn_cancel = QPushButton("Отмена")
537          btn_cancel.clicked.connect(dialog.reject)
538          btn_layout.addWidget(btn_ok)
539          btn_layout.addWidget(btn_cancel)
540          layout.addLayout(btn_layout)
541  
542          dialog.exec()

544      def do_export(self, dialog, start, end, export_func, default_filename):
545          path, _ = QFileDialog.getSaveFileName(self, "Сохранить файл", default_filename, "Excel (*.xlsx)")
546          if path:
547              try:
548                  export_func(start.toString("yyyy-MM-dd"), end.toString("yyyy-MM-dd"), path)
549                  QMessageBox.information(self, "Успех", "Отчёт успешно сохранён!")
550                  dialog.accept()
551              except Exception as e:
552                  QMessageBox.critical(self, "Ошибка", str(e))
