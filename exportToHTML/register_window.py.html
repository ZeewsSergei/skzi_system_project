42       def handle_register(self):
43           user = self.username.text().strip()
44           pwd = self.password.text()
45           conf = self.confirm.text()
46           role = self.role.currentText()
47           if not user or not pwd:
48               QMessageBox.warning(self, "Ошибка", "Заполните все поля")
49               return
50           if pwd != conf:
51               QMessageBox.warning(self, "Ошибка", "Пароли не совпадают")
52               return
53           if len(pwd) < 6:
54               QMessageBox.warning(self, "Ошибка", "Пароль должен быть не менее 6 символов")
55               return
56           if self.auth.create_user(user, pwd, role):
57               QMessageBox.information(self, "Успех", f"Пользователь {user} создан")
58               self.close()
59           else:
60               QMessageBox.critical(self, "Ошибка", "Пользователь с таким именем уже существует")
