40       def refresh(self):
41           rows = get_all_trainings()
42           self.table.setRowCount(0)
43           for i, row in enumerate(rows):
44               self.table.insertRow(i)
45               for j, val in enumerate(row):
46                   self.table.setItem(i, j, QTableWidgetItem(str(val) if val else ""))
47           self.filter_table()  # применить текущий поиск
48   
49       def filter_table(self):
50           text = self.search.text().lower()
51           for i in range(self.table.rowCount()):
52               match = False
53               for j in range(self.table.columnCount()):
54                   item = self.table.item(i, j)
55                   if item and text in item.text().lower():
56                       match = True
57                       break
58               self.table.setRowHidden(i, not match)
