1    from PyQt6.QtWidgets import QDialog, QVBoxLayout, QLabel, QLineEdit, QPushButton, QMessageBox
2    from PyQt6.QtCore import Qt
3    from security.auth_service import AuthService
4    from main_window import MainWindow
5    from forms.change_password_dialog import ChangePasswordDialog
6    
7    class LoginWindow(QDialog):
8        def __init__(self):
9            super().__init__()
10           self.change_btn = None
11           self.login_btn = None
12           self.pass_input = None
13           self.user_input = None
14           self.auth = AuthService()
15           self.main_win = None
16           self.init_ui()
17   
18       def init_ui(self):
19           self.setWindowTitle("Авторизация - Учет СКЗИ")
20           self.setFixedSize(350, 250)
21           layout = QVBoxLayout()
22           layout.setSpacing(15)
23   
24           header = QLabel("Вход в систему")
25           header.setAlignment(Qt.AlignmentFlag.AlignCenter)
26           header.setStyleSheet("font-size: 18px; font-weight: bold;")
27           layout.addWidget(header)
28   
29           self.user_input = QLineEdit(placeholderText="Логин")
30           self.user_input.returnPressed.connect(self.handle_login)
31           layout.addWidget(self.user_input)
32   
33           self.pass_input = QLineEdit(placeholderText="Пароль", echoMode=QLineEdit.EchoMode.Password)
34           self.pass_input.returnPressed.connect(self.handle_login)
35           layout.addWidget(self.pass_input)
36   
37           self.login_btn = QPushButton("Войти")
38           self.login_btn.setObjectName("primaryButton")
39           self.login_btn.clicked.connect(self.handle_login)
40           layout.addWidget(self.login_btn)
41   
42           self.change_btn = QPushButton("Сменить пароль")
43           self.change_btn.setStyleSheet("color: #7f8c8d; border: none; background: none; text-decoration: underline;")
44           self.change_btn.clicked.connect(self.open_change_password)
45           layout.addWidget(self.change_btn)
46   
47           self.setLayout(layout)
48   
49       def handle_login(self):
50           username = self.user_input.text().strip()
51           password = self.pass_input.text().strip()
52           if not username or not password:
53               QMessageBox.warning(self, "Ошибка", "Введите логин и пароль")
54               return
55           user = self.auth.authenticate(username, password)
56           if user:
57               self.main_win = MainWindow(user)
58               self.main_win.show()
59               self.accept()
60           else:
61               QMessageBox.critical(self, "Ошибка", "Неверный логин или пароль")
62   
63       def open_change_password(self):
64           dlg = ChangePasswordDialog(parent=self)
65           dlg.exec()
