72   def get_icon_path():
73       app_root = get_app_root()
74       candidates = [
75           os.path.join(app_root, 'resources', 'app_icon.ico'),
76           os.path.join(app_root, 'app_icon.ico'),
77           os.path.join(app_root, 'icon.ico'),
78       ]
79       for path in candidates:
80           if os.path.exists(path):
81               return path
82       return None

84   def global_exception_handler(exctype, value, tb):

87       try:
88           QMessageBox.critical(None, "Критическая ошибка",
89                                f"Произошла непредвиденная ошибка:\n{str(value)}\n\nПодробности в консоли.")
90       except:
91           pass
92       sys.exit(1)
